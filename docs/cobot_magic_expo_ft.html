# client/envs/cobot_env.py
"""Cobot Magic single-arm env. Same RPC surface as DroidEnv for run_client.py."""
import numpy as np
from openpi_client import image_tools
from client.real_utils.detector import success_detector_manual
from examples.mobile_aloha_AgileX.real_env import make_real_env


class CobotMagicEnv:
    def __init__(
        self,
        language_instruction,
        image_size=(180, 320),
        control_hz=30,
        auto_reset_steps=400,
        video_dir=None,
        use_single_arm=True,
        **kwargs,
    ):
        self.language_instruction = language_instruction
        self.image_size = tuple(image_size)
        self.control_hz = control_hz
        self.auto_reset_steps = int(auto_reset_steps)
        self.video_dir = video_dir
        self._env = make_real_env(
            init_node=True,
            control_freq_hz=control_hz,
            use_single_arm=use_single_arm,   # real_env.SINGLE_ARM_NAME == "right"
        )
        self._steps = 0
        self.prev_obs = None

    def reset(self):
        self._steps = 0
        self._env.reset()
        return self.get_observation()

    def _pack(self, raw):
        def _img(x):
            x = np.asarray(x)
            if x.ndim == 3 and x.shape[0] == 3:
                x = np.transpose(x, (1, 2, 0))
            return image_tools.convert_to_uint8(
                image_tools.resize_with_pad(x, self.image_size[0], self.image_size[1])
            )
        imgs = raw["images"]
        qpos = np.asarray(raw["qpos"], np.float32).reshape(-1)
        if qpos.shape[0] != 7:
            raise ValueError(f"single-arm qpos must be 7D, got {qpos.shape}")
        return {
            "observation.images.cam_high": _img(imgs["cam_high"]),
            "observation.images.cam_right_wrist": _img(imgs["cam_right_wrist"]),
            "observation.state": qpos,
            "prompt": self.language_instruction,
        }

    def get_observation(self):
        raw = self._env.get_observation()
        self.prev_obs = raw
        return self._pack(raw)

    def step(self, action):
        self._steps += 1
        action = np.asarray(action, np.float64).reshape(-1)[:7]
        self._env.step(action)   # 绝对关节；gripper 偏置已在 real_env 内 -0.005
        return {"executed_action": action}

    def get_info_for_step(self, raw_obs=None):
        time_stop = self._steps >= self.auto_reset_steps
        manual = success_detector_manual()   # 一行：s 回车=成功，f 回车=失败，q 回车=作废继续
        success = manual == "success"
        terminate = manual == "reset" or time_stop
        done = bool(success or terminate)
        return done, success, (1.0 if success else 0.0), (0.0 if done else 1.0)

    def render(self, *args, **kwargs):
        return None

4.3 expo_ft/env/cobot_utils.py 读 LeRobot

实现已在仓库里。datapath 是含 meta/info.json 的数据集根目录。H20 上 offline_train.sh 的默认集是 $HF_LEROBOT_HOME/screw_sorting_single_sft_ep279_annotated（279 段，115130 帧）。相机特征的 dtype 是 image，图像在 parquet 里。OpenPI 的 LeRobotDataset.__getitem__ 也是这条路径：只有 dtype=video 才会去打开 mp4。

旁边的 videos/chunk-000/cam_high/ 和 cam_right_wrist/ 不是训练图。第 66 段 mp4 有 451 帧，parquet 只有 417 帧，对不上。loader 按 episode 并行读 parquet，再缩到 224。环境变量 COBOT_LOAD_WORKERS 默认 8。进度条 cobot lerobot 的单位是 episode，不是帧。hdf5 只是转 LeRobot 之前的原始录制，这条训练不吃它。

OpenPI 的 scripts/train.py 用 num_workers=8 的 DataLoader，训练走一步才解码当前 batch，所以索引建完就能出 step。这边的回放缓冲区要先装进全部帧，cobot lerobot 走完才会开始梯度。这是启动时间的差别，不是又换了一套图像。

冻住 π0.5 时有什么损失

对当前「只训残差与 Q」的路径，算法上没有额外损失：本来就不打算改 π0.5。代价是：

  • 不能再做在线前缀行为克隆；若以后改回 train_base_actor=True，需要足够显存重新建 Adam。
  • 权重以 bf16 驻留（与 openpi 正式推理一致），相对 fp32 推理通常可忽略。
  • 不保留 EMA 副本；冻住时 EMA 本来也不会更新。

数据怎么进训练

开训时整份 LeRobot 过 Cobot 的 repack、关节 delta、下载来的 norm，插进一个 PiReplayBuffer，标成成功演示。真机每一拍先排队，回合结束才写入：IPC 终端输入 s 并回车为成功，输入 f 并回车或步数到 auto_reset_steps 为失败。不会再写成一份新的 LeRobot。更新在 10 个回合之后、并且样本不少于 batch 时解开；回合结束时按大约每 30 拍一次梯度来补。

终端标注

标注只在运行 run_policy.sh 的 IPC 终端里生效。GPU 上运行 eval_rtc_sft.sh 或 run_server.sh 的终端按键无效。

每次 reset 先把机械臂回零，然后停住，直到同一终端按 o 再回车。这段时间用来摆螺丝、整理场景。按 o 之后策略才发第一帧动作。训练和评测的每个情节都要按一次 o，包括第一情节和超时失败之后。

情节进行中，CobotMagicEnv.get_info_for_step 每一拍调用 success_detector_manual。该函数对 /dev/tty 做超时为 0 的 select。终端里没有待读的一整行时立刻返回，控制环继续，手臂不停。读入使用 readline，因此必须是字母后接回车构成的一行。只按字母、不按回车时，这一行尚未结束；下一拍发现终端已有字符，就会停在 readline 里等待回车。等待期间控制环停在这一拍，奖励也尚未写入。

表　IPC 终端一行输入的含义

输入何时结果
o 然后回车回零之后、情节开始前启动情节。其它键不会开始。
s 然后回车情节进行中成功。reward=1，情节结束，随后回零，再等下一次 o。
f 然后回车情节进行中失败。reward=0，情节结束，随后回零，再等下一次 o。
q 然后回车情节进行中这次按键作废，情节继续。
没有输入情节进行中继续执行，直到步数达到 auto_reset_steps，按失败回零，再等 o。

奖励记在读到 s 或 f 这一整行的那一拍。中间各拍奖励为 0。相对螺丝已经放好的时刻，滞后主要来自操作者看到结果再输入。回车是这一行的结束符。eval_rtc_sft.sh 不用该奖励更新网络，只决定日志里这一回合是否成功。

采用按行读取，是为了让 30 Hz 控制环在操作者提交之前不查询键盘，也避免把误触的单个字符当成回合结束。

1. RTC-SFT（H20，train_step_p1_prefix）

实现在 expo_ft/agents/vla/pi05.py 的 train_step_p1_prefix。expo_ft/agents/alg/rtc.py 的 update_actor 在 p1_use_prefix_conditioning 为真时调用它。一条样本是观测 o 和干净动作块 a∈R^{H×D}，D=32 是 padding 后的动作维。每个样本抽

d ~ Unif{0, 1, …, 10}
τ_b ~ 0.001 + 0.999 · Beta(1.5, 1)
m_h = 1[h ≥ d]
τ_h = τ_b · m_h
x_h = τ_h ε_h + (1 − τ_h) a_h
u_h = ε_h − a_h

h<d 时 τ_h=0，所以 x_h=a_h，prefix 是干净动作。网络输出速度 v_θ(o, x, τ)。损失只加在 postfix 上：

L_RTC = ( Σ_{b,h} m_h · mean_D (v_θ − u)² ) / Σ_{b,h} m_h

d=0 时整段都进损失，就是普通 flow-matching BC。在线脚本当前不调用该损失（train_base_actor=False）。若打开，则 d 不再随机：回合前 r 步用 d=0，之后用部署的 d。

2. 真机一步怎么选出要执行的窗

队列还剩 d=6 拍时，用已经发出去的这 6 拍做 prefix，pi0.5 采 N=8 条整段，再切执行窗

w_i = chunk_i[d : d+r] ∈ R^{10×7},   i = 1…8

edit actor 对每条基座采残差 ρ_i，乘上系数后加回去（edit_scale=0.1）：

ã_i = w_i + 0.1 · ρ_i,   ρ_i ~ π_edit(· | enc(o), s, w_i)

候选集有 16 条，都在 R^{70}：

C = { w_1, …, w_8, ã_1, …, ã_8 }

10 个 Q 里随机抽 num_min_qs=2 个，取得分最小的那个，再取最大：

Q̄(o, C_k) = min_{j∈J} Q_j(enc(o), C_k, s),   |J|=2
k* = argmax_k Q̄(o, C_k)

手臂执行 C_{k*} 的 10 拍。这 10 拍的最后 6 拍成为下一次的 prefix。

3. 训练时 Q、Q_f、edit、基座各自的损失

训练 backup 不去噪 8 条。先抽 filter_N=8 个噪声种子 ε_1…ε_8，Q_f 同时看当前观测和 delay 之前那一帧（filter_add_delayed_obs=True）。filter_temperature=0，所以

ε* = argmax_n Q_f(o', o_{t−d}, ε_n)

只用 ε* 去噪出一条基座 w，再加 filter_n_edit=1 条残差 ã。TD 用的下一动作是这两条里 Q 更高的那条 a*。n-step 回报已经在 replay 里按窗加好（Cobot 中间奖励为 0，只有结束那一拍是 1 或 0）：

R = Σ_{i=0}^{r−1} γ^i r_{t+i}
y = R + γ^r · mask · min_{j∈J} Q'_j(o', a*)
L_Q = mean( valids · (Q(o, a_buf) − y)² )

γ^10 = 0.99^10 ≈ 0.904。超时截断的窗 valids=0，不进这个均值；回合自然结束但没走满执行窗的窗因为 valids_keep_terminal_windows=True 仍保留。

Q_f 的回归目标是基座那条的 Q，不是 argmax 之后的那条：

L_f = mean ( Q_f(o', ε*) − stopgrad(Q̄(o', w)) )²

edit actor 是在 buffer 里的基座动作上做的。残差乘 0.1 之后，对数概率要减掉雅可比：

a = a_buf + 0.1 · ρ
log π_scaled(a) = log π_edit(ρ) − 70 · log(0.1)
L_edit = mean( α · log π_scaled(a) − Q(o, a) )

70·log(0.1) ≈ −161.2。温度 α 把熵拉向 H* = −70/2 = −35（adjust_target_entropy=False，不把 log(edit_scale) 再加进目标熵）。当前脚本 train_base_actor=False，因此在线阶段不再对 π0.5 做 prefix BC；基座权重保持 RTC-SFT 检查点，只更新 critic、Q_f 与 edit。
